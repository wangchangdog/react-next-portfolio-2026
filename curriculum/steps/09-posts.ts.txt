import "server-only";
import type { MicroCMSImage, MicroCMSListContent } from "microcms-js-sdk";
import { getMicroCMSClient } from "@/lib/microcms";
import { samplePosts } from "@/data/sample-posts";
import type { Post } from "@/types/post";

type CMSPost = {
  title: string;
  description?: string;
  content?: string;
  thumbnail?: MicroCMSImage;
} & MicroCMSListContent;

function toPost(item: CMSPost): Post {
  const image = item.thumbnail;
  const thumbnail = image && image.width && image.height
    ? { url: image.url, width: image.width, height: image.height }
    : undefined;
  return {
    id: item.id,
    slug: item.id,
    title: item.title,
    description: item.description ?? "",
    content: item.content ?? "",
    publishedAt: item.publishedAt ?? item.createdAt,
    thumbnail,
  };
}

// 標準課題は最大100件。100件を超える場合はページネーション課題へ進みます。
export async function getPosts(): Promise<Post[]> {
  const client = getMicroCMSClient();
  const response = await client.getList<CMSPost>({
    endpoint: "blogs",
    queries: { limit: 100, orders: "-publishedAt" },
    customRequestInit: { cache: "no-store" },
  });
  return response.contents.map(toPost);
}

// この段階の詳細はサンプルだけです。CMS記事の詳細はT10で接続します。
export async function getPostBySlug(slug: string): Promise<Post | undefined> {
  return samplePosts.find((item) => item.slug === slug);
}
