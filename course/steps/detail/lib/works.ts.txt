import type { MicroCMSListContent } from "microcms-js-sdk";
import { sampleWorks } from "@/data/sample-works";
import { getMicroCMSClient } from "@/lib/microcms";
import { isCMSNotFound } from "@/lib/cms-errors";
import type { Work } from "@/types/work";
import type { Thumbnail } from "@/types/media";

type CMSWork = {
  title: string;
  description?: string;
  content?: string;
  thumbnail?: Thumbnail;
} & MicroCMSListContent;

function toWork(work: CMSWork): Work {
  return {
    id: work.id,
    slug: work.id,
    title: work.title,
    description: work.description ?? "",
    content: work.content ?? "",
    thumbnail: work.thumbnail,
  };
}

export async function getWorks(): Promise<Work[]> {
  const client = getMicroCMSClient();
  if (!client) return [...sampleWorks];
  const data = await client.getList<CMSWork>({
    endpoint: "works",
    queries: { limit: 10, orders: "-publishedAt" },
    customRequestInit: { cache: "no-store" },
  });
  return data.contents.map(toWork);
}

export async function getWorkBySlug(slug: string): Promise<Work | undefined> {
  const client = getMicroCMSClient();
  if (!client) return sampleWorks.find((work) => work.slug === slug);
  try {
    const work = await client.getListDetail<CMSWork>({
      endpoint: "works",
      contentId: encodeURIComponent(slug),
      customRequestInit: { cache: "no-store" },
    });
    return toWork(work);
  } catch (error) {
    if (!isCMSNotFound(error)) throw error;
    await client.getList<CMSWork>({
      endpoint: "works",
      queries: { limit: 1, fields: "id" },
      customRequestInit: { cache: "no-store" },
    });
    return undefined;
  }
}
